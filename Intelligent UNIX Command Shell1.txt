#ifndef PROG04_H
#define PROG04_H

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_FRAMES 1024

struct page
{
    int page_no;
    int ref_count;
    int dirty;
    int valid;
};

#endif
----------------------------------------------------------------------------
prog.c
----------------------------------------------------------------------------



#include "prog04.h"

struct page memory[MAX_FRAMES];

int frames;
int total_accesses = 0;
int page_faults = 0;
int pages_written = 0;

int fifo_position = 0;


/* Reduce reference count of all pages */
void reduce_reference_count()
{
    int i;

    for(i = 0; i < frames; i++)
    {
        if(memory[i].valid == 1)
        {
            if(memory[i].ref_count > 0)
                memory[i].ref_count--;
        }
    }
}


/* Find page in physical memory */
int find_page(int page_no)
{
    int i;

    for(i = 0; i < frames; i++)
    {
        if(memory[i].valid == 1 &&
           memory[i].page_no == page_no)
        {
            return i;
        }
    }

    return -1;
}


/*
 * Find a page whose reference count is 0.
 * FIFO is followed starting from fifo_position.
 */
int find_victim()
{
    int i;
    int pos;

    while(1)
    {
        for(i = 0; i < frames; i++)
        {
            pos = (fifo_position + i) % frames;

            if(memory[pos].valid == 1 &&
               memory[pos].ref_count == 0)
            {
                return pos;
            }
        }

        /*
         * No page has reference count 0.
         * Reduce all counts and try again.
         */
        reduce_reference_count();
    }
}


/* Load a page into an empty frame */
void load_empty_page(int frame,
                     int page_no,
                     char operation,
                     int debug)
{
    memory[frame].page_no = page_no;

    /*
     * Initial reference count is 3.
     * Initial access does NOT increase it.
     */
    memory[frame].ref_count = 3;

    memory[frame].valid = 1;

    if(operation == 'w')
        memory[frame].dirty = 1;
    else
        memory[frame].dirty = 0;

    if(debug)
    {
        printf("Page NULL replaced by Page %d\n",
               page_no);
    }
}


/* Replace an existing page */
void replace_page(int frame,
                  int page_no,
                  char operation,
                  int debug)
{
    int old_page;
    int old_dirty;

    old_page = memory[frame].page_no;
    old_dirty = memory[frame].dirty;


    if(debug)
    {
        printf("Page %d replaced by Page %d\n",
               old_page,
               page_no);

        if(old_dirty)
            printf("Page %d was dirty\n",
                   old_page);
        else
            printf("Page %d was not dirty\n",
                   old_page);
    }


    if(old_dirty)
        pages_written++;


    memory[frame].page_no = page_no;

    memory[frame].ref_count = 3;

    memory[frame].valid = 1;


    if(operation == 'w')
        memory[frame].dirty = 1;
    else
        memory[frame].dirty = 0;


    /*
     * The new page becomes the newest page
     * in FIFO order.
     */
    fifo_position = (frame + 1) % frames;
}


/* Main */
int main(int argc, char *argv[])
{
    int virtual_memory;
    int page_size;
    int physical_memory;

    int virtual_bits;
    int page_bits;

    char *input_file;
    char debug_option;

    FILE *fp;

    unsigned int address;
    char operation;

    int page_no;
    int frame;

    int i;


    /*
     * Check command line arguments
     *
     * prog04 VM PageSize PhysicalMemory input-file d/n
     */
    if(argc != 6)
    {
        printf("Usage: %s <VirtualMemory> <PageSize> <PhysicalMemory> <input-file> d/n\n",
               argv[0]);

        return 1;
    }


    virtual_bits = atoi(argv[1]);
    page_bits = atoi(argv[2]);
    physical_memory = atoi(argv[3]);

    input_file = argv[4];
    debug_option = argv[5][0];


    if(debug_option != 'd' &&
       debug_option != 'n')
    {
        printf("Last argument must be d or n\n");
        return 1;
    }


    /*
     * Virtual memory and page size are given
     * as powers of 2.
     */
    virtual_memory = 1 << virtual_bits;
    page_size = 1 << page_bits;

    frames = physical_memory;


    if(frames > MAX_FRAMES)
    {
        printf("Too many physical memory frames\n");
        return 1;
    }


    if(virtual_bits > 20)
    {
        printf("Virtual memory cannot exceed 2^20\n");
        return 1;
    }


    if(page_bits < 10)
    {
        printf("Page size must be at least 2^10\n");
        return 1;
    }


    /* Open input file */
    fp = fopen(input_file, "r");

    if(fp == NULL)
    {
        perror("fopen");
        return 1;
    }


    /*
     * Initialize physical memory.
     */
    for(i = 0; i < frames; i++)
    {
        memory[i].valid = 0;
        memory[i].page_no = -1;
        memory[i].ref_count = 0;
        memory[i].dirty = 0;
    }


    fifo_position = 0;


    if(debug_option == 'd')
        printf("// Simulation Starts\n");


    /*
     * Read every memory access.
     */
    while(fscanf(fp, "%u %c",
                  &address,
                  &operation) == 2)
    {
        total_accesses++;


        /*
         * Convert physical address to page number.
         *
         * Example:
         *
         * page size = 2^12 = 4096
         *
         * page number = address / 4096
         */
        page_no = address / page_size;


        /*
         * Check whether page is already
         * present in physical memory.
         */
        frame = find_page(page_no);


        if(frame != -1)
        {
            /*
             * Page hit.
             *
             * Increase reference count.
             * Maximum = 10.
             */
            if(memory[frame].ref_count < 10)
                memory[frame].ref_count++;


            /*
             * Write operation makes page dirty.
             */
            if(operation == 'w')
                memory[frame].dirty = 1;
        }
        else
        {
            /*
             * Page is not present.
             */

            /*
             * First find an empty frame.
             */
            frame = -1;

            for(i = 0; i < frames; i++)
            {
                if(memory[i].valid == 0)
                {
                    frame = i;
                    break;
                }
            }


            if(frame != -1)
            {
                /*
                 * Empty frame.
                 *
                 * This is NOT counted as
                 * a page fault according
                 * to the assignment.
                 */
                load_empty_page(frame,
                                 page_no,
                                 operation,
                                 debug_option == 'd');


                /*
                 * Move FIFO position forward.
                 */
                fifo_position = (frame + 1) % frames;
            }
            else
            {
                /*
                 * Physical memory is full.
                 *
                 * Find a page with reference
                 * count 0 using FIFO.
                 */
                frame = find_victim();


                /*
                 * This is a page fault because
                 * an existing page is replaced.
                 */
                page_faults++;


                replace_page(frame,
                             page_no,
                             operation,
                             debug_option == 'd');
            }
        }


        /*
         * After every 4 memory accesses,
         * reduce reference count by 1.
         */
        if(total_accesses % 4 == 0)
        {
            reduce_reference_count();
        }
    }


    fclose(fp);


    if(debug_option == 'd')
        printf("// Simulation Ends\n");


    printf("\n// Display Statistics\n");

    printf("Number of memory accesses = %d\n",
           total_accesses);

    printf("Number of memory accesses that resulted in page faults = %d\n",
           page_faults);

    printf("Number of pages written to disk = %d\n",
           pages_written);


    return 0;
}
